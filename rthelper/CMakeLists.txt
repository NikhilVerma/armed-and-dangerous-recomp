cmake_minimum_required(VERSION 3.24)
project(rt_helper LANGUAGES C CXX)

# The 64-bit ray-tracing helper (src/rhi/vulkan/rt_ipc.h): the game is 32-bit and NVIDIA's 32-bit Vulkan driver has
# no ray tracing, so the RT cores are used from this separate 64-bit process. Build: tools/build_helper.ps1.
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Configure from an x64 developer shell; the helper is 64-bit.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(FetchContent)
# The same pinned Khronos sources as the hook (hook/CMakeLists.txt).
FetchContent_Declare(vulkan_headers
  GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git
  GIT_TAG vulkan-sdk-1.3.296.0)
FetchContent_Declare(volk
  GIT_REPOSITORY https://github.com/zeux/volk.git
  GIT_TAG vulkan-sdk-1.3.296.0)
FetchContent_Declare(glslang
  GIT_REPOSITORY https://github.com/KhronosGroup/glslang.git
  GIT_TAG vulkan-sdk-1.3.296.0)
set(VOLK_PULL_IN_VULKAN OFF CACHE BOOL "" FORCE)
set(VOLK_HEADERS_ONLY ON CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(vulkan_headers volk glslang)

# AMD FidelityFX Shadow Denoiser (third_party/FidelityFX-ShadowDenoiser, MIT, unmodified), embedded by file name for
# run-time compilation (shadow_denoiser.cpp FfxSource).
set(FFX_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/FidelityFX-ShadowDenoiser)
file(GLOB_RECURSE FFX_FILES CONFIGURE_DEPENDS ${FFX_DIR}/*.h ${FFX_DIR}/*.glsl)
set(FFX_EMBED ${CMAKE_BINARY_DIR}/generated/ffx_embed.inc)
set(FFX_TABLE ${CMAKE_BINARY_DIR}/generated/ffx_table.inc)
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/generated)
file(WRITE ${FFX_EMBED} "// Generated by rthelper/CMakeLists.txt.\n")
file(WRITE ${FFX_TABLE} "// Generated by rthelper/CMakeLists.txt.\n")
foreach(f ${FFX_FILES})
  get_filename_component(n ${f} NAME)
  file(READ ${f} hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
  string(MAKE_C_IDENTIFIER ${n} id)
  file(APPEND ${FFX_EMBED} "static const char k_${id}[] = {${bytes}0};\n")
  file(APPEND ${FFX_TABLE} "{\"${n}\", k_${id}},\n")
endforeach()

add_executable(rt_helper WIN32 main.cpp shadow_denoiser.cpp ffx_embed.cpp aa.cpp)
target_include_directories(rt_helper PRIVATE ${volk_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../src/rhi/vulkan
  ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/ffx-api/include
  ${CMAKE_BINARY_DIR}/generated)
target_compile_definitions(rt_helper PRIVATE VK_NO_PROTOTYPES VK_USE_PLATFORM_WIN32_KHR)
target_link_libraries(rt_helper PRIVATE Vulkan::Headers glslang::glslang glslang::SPIRV
  glslang::glslang-default-resource-limits)
target_compile_options(rt_helper PRIVATE /W4 /permissive- /utf-8)

# Temporal anti-aliasing (aa.cpp, docs/subsystems/remaster-aa.md): AMD's prebuilt, signed FidelityFX API DLL for
# Vulkan (FSR 3.1) from the FidelityFX SDK v1.1.4 (MIT), downloaded once and placed next to rt_helper.exe.
set(FFX_VK_DLL ${CMAKE_BINARY_DIR}/amd_fidelityfx_vk.dll)
if(NOT EXISTS ${FFX_VK_DLL})
  file(DOWNLOAD
    https://github.com/GPUOpen-LibrariesAndSDKs/FidelityFX-SDK/raw/v1.1.4/PrebuiltSignedDLL/amd_fidelityfx_vk.dll
    ${FFX_VK_DLL}
    EXPECTED_HASH SHA256=a1624cc4238fef046f30c4d80ce3f47be63fc5f5373f49e3ee9edb9960f54c78
    SHOW_PROGRESS)
endif()
